public class StudentService {

    // Defined ONCE as a reusable constant or class variable
    public static final Comparator<Student> CUSTOM_COMPARATOR = Comparator
        .comparingInt(Student::getScore).reversed() // LOW TO HIGH
        .thenComparing((a, b) -> {
            if (a.getStatus() == Status.ACTIVE && b.getStatus() != Status.ACTIVE) return -1;
            if (a.getStatus() != Status.ACTIVE && b.getStatus() == Status.ACTIVE) return 1;
            return 0;
        })
        .thenComparing(Student::getName);

    public void processStudents(List<Student> students, Student s1, Student s2) {
        
        // 1. Use it for batch sorting a collection
        students.sort(CUSTOM_COMPARATOR);

        // 2. Use it for a single comparison in an if-statement
        if (CUSTOM_COMPARATOR.compare(s1, s2) < 0) {
            System.out.println(s1.getName() + " takes priority over " + s2.getName());
        }
    }
}
